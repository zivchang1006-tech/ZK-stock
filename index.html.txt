  <script>
    // 自動去讀取同一資料夾下的 data.json 檔案
    fetch('data.json?t=' + new Date().getTime())
      .then(response => response.json())
      .then(history => {
        if (history.length === 0) {
          document.getElementById('update-time').innerText = "目前尚無數據，等待機器人更新";
          return;
        }
        const latest = history;
        document.getElementById('update-time').innerText = "最後自動更新：" + latest.update_time;

        const cards = [
          { code: "05 三大法人", title: "外資現貨(億)", val: latest.foreign },
          { code: "06 三大法人", title: "投信現貨(億)", val: latest.trust },
          { code: "07 三大法人", title: "自營商現貨(億)", val: latest.dealer }
        ];

        document.getElementById('cards-container').innerHTML = cards.map(c => {
          let isRed = c.val > 0;
          return `
            <div class="card">
              <div class="card-header"><span>${c.code}</span><i class="dot ${isRed ? 'red' : 'green'}"></i></div>
              <div style="color:#b0bac5; font-size:13px;">${c.title}</div>
              <div class="card-value ${isRed ? 'text-red' : 'text-green'}">${c.val}</div>
            </div>
          `;
        }).join('');

        document.getElementById('history-body').innerHTML = history.map(row => {
          let total = (row.foreign + row.trust + row.dealer).toFixed(2);
          return `
            <tr>
              <td>${row.date}</td>
              <td class="${row.foreign > 0 ? 'text-red' : 'text-green'}">${row.foreign}</td>
              <td class="${row.trust > 0 ? 'text-red' : 'text-green'}">${row.trust}</td>
              <td class="${row.dealer > 0 ? 'text-red' : 'text-green'}">${row.dealer}</td>
              <td class="${total > 0 ? 'text-red' : 'text-green'}"><b>${total}</b></td>
            </tr>
          `;
        }).join('');
      });
  </script>
